Short answer: almost! Your coroutine never starts, and a few names/assumptions can bite you (e.g., audio shadows Unity’s obsolete Component.audio, canvas might be null, and the Animator may live on a child).

Here’s a tightened, drop-in version that fixes those issues and keeps your API the same:

csharp
using System.Collections;
using UnityEngine;
using UnityEngine.UI;

public abstract class NarrativeObjects : MonoBehaviour
{
    [HideInInspector] public AudioSource audioSource;
    [HideInInspector] public ParticleSystem particles;
    [HideInInspector] public Canvas canvas;
    [HideInInspector] public Image[] images;
    [HideInInspector] public Animator animator;

    // Use protected virtual so subclasses can override while still calling base.Start()
    protected virtual void Start()
    {
        // Try to find components on self first, then on children as a fallback where sensible.
        audioSource = GetComponent<AudioSource>();
        if (!audioSource) audioSource = GetComponentInChildren<AudioSource>();

        particles = GetComponentInChildren<ParticleSystem>();

        canvas = GetComponentInChildren<Canvas>();
        images = (canvas != null) ? canvas.GetComponentsInChildren<Image>(true) : System.Array.Empty<Image>();

        animator = GetComponent<Animator>();
        if (!animator) animator = GetComponentInChildren<Animator>();
    }

    protected virtual void Update() { }

    public virtual void Signpost()
    {
        if (audioSource) audioSource.Play();
        if (particles) particles.Play();
        Debug.Log($"{gameObject.name} has started signposting!");
    }

    public virtual void UnSignpost()
    {
        if (audioSource) audioSource.Stop();
        if (particles) particles.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);
        Debug.Log($"{gameObject.name} has stopped signposting!");
    }

    public virtual void PlayAnimation()
    {
        // IMPORTANT: actually start the coroutine
        StartCoroutine(StartAnimation());
    }

    private IEnumerator StartAnimation()
    {
        if (animator)
        {
            animator.SetBool("hasTouched", true); // make sure these parameters exist in your Animator Controller
            yield return new WaitForSeconds(10f);
            animator.SetBool("hasRead", true);
        }
        else
        {
            Debug.LogWarning($"[{name}] Animator not found. Skipping animation booleans.");
        }
    }

    protected virtual void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Hand"))
        {
            PlayAnimation();
        }
    }
}

Why these changes:

  • StartCoroutine(StartAnimation()); — your original called the IEnumerator like a normal method, so it never yielded/timed.

  • Renamed audio → audioSource to avoid shadowing Unity’s legacy Component.audio.

  • Added null-safety for canvas, particles, audioSource, and animator.

  • Look for Animator (and AudioSource) on self and children; many setups keep them on child objects.

  • Made lifecycle methods protected virtual so subclasses can override cleanly and still call base.Start().

Checklist to ensure it runs:

  • The GameObject (or a child) has an Animator with bool parameters hasTouched and hasRead.

  • This object’s Collider is isTrigger = true, and either this object or the “Hand” object has a Rigidbody.

  • The entering collider has tag “Hand”.

  • If you need the animation to restart later, remember to reset those bools to false somewhere appropriate.

other.transform.position is the Hand object’s origin, not the actual contact point. Triggers don’t give contact points, so you need to approximate it.

Here’s a safe drop-in tweak that spawns your VFX where the colliders meet:

csharp
using UnityEngine;
using UnityEngine.UI;
using System.Collections;

public abstract class NarrativeObjects : MonoBehaviour
{
    [HideInInspector] public AudioSource audioSource;
    [HideInInspector] public ParticleSystem particles;
    [HideInInspector] public Canvas canvas;
    [HideInInspector] public Image[] images;
    [HideInInspector] public Animator animator;

    [Header("VFX")]
    public GameObject touchVFX;

    private Collider myCol;

    protected virtual void Start()
    {
        audioSource = GetComponent<AudioSource>() ?? GetComponentInChildren<AudioSource>();
        particles = GetComponentInChildren<ParticleSystem>();
        canvas = GetComponentInChildren<Canvas>();
        images = canvas ? canvas.GetComponentsInChildren<Image>(true) : System.Array.Empty<Image>();
        animator = GetComponent<Animator>() ?? GetComponentInChildren<Animator>();
        myCol = GetComponent<Collider>() ?? GetComponentInChildren<Collider>();
    }

    public virtual void PlayAnimation(Vector3 touchLocation)
    {
        StartCoroutine(StartAnimation(touchLocation));
    }

    private IEnumerator StartAnimation(Vector3 touchLocation)
    {
        if (touchVFX) Instantiate(touchVFX, touchLocation, Quaternion.identity);

        if (particles)
            yield return new WaitWhile(() => particles.isPlaying);
        else
            yield return null;

        if (animator)
        {
            animator.SetBool("hasTouched", true);
            yield return new WaitForSeconds(10f);
            animator.SetBool("hasRead", true);
        }
    }

    protected virtual void OnTriggerEnter(Collider other)
    {
        if (!other.CompareTag("Hand")) return;

        UnSignpost();

        // Approximate contact point for TRIGGERS:
        Vector3 pointOnMe   = myCol ? myCol.ClosestPoint(other.bounds.center) : transform.position;
        Vector3 pointOnThem = other.ClosestPoint(pointOnMe);
        Vector3 spawnPos    = (pointOnMe + pointOnThem) * 0.5f;

        PlayAnimation(spawnPos);
    }

    // Your existing signpost methods…
    public virtual void Signpost()   { if (audioSource) audioSource.Play(); if (particles) particles.Play(); }
    public virtual void UnSignpost() { if (audioSource) audioSource.Stop(); if (particles) particles.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear); }
}

Notes:

  • Collider.ClosestPoint(...) gives you a point on each collider’s surface; averaging them is a good approximation for the “touch” spot.

  • If you need the true contact point and normal, switch to non-trigger physics (OnCollisionEnter(Collision c)) and use c.contacts[0].point / .normal.

  • Optional: orient the VFX to the impact “normal”:

    csharp
    Vector3 normal = (pointOnMe - pointOnThem).normalized;
Quaternion rot = normal.sqrMagnitude > 1e-6f ? Quaternion.LookRotation(normal) : Quaternion.identity;
Instantiate(touchVFX, spawnPos, rot);

Make sure at least one of the two colliding objects has a Rigidbody (kinematic is fine) so trigger events fire reliably.
